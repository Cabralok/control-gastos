function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = sheet.getDataRange().getValues(); let incomes = []; let expenses = []; for (let i = 3; i < data.length; i++) { const rowNumber = i + 1; // Número real de fila en Google Sheets const estado = data[i][2]; // Columna C const conceptoGasto = data[i][3]; // Columna D const montoGasto = data[i][4]; // Columna E const conceptoIngreso = data[i][5];// Columna F const montoIngreso = data[i][6]; // Columna G if (conceptoGasto && conceptoGasto.toString().trim().toLowerCase() !== 'total' && montoGasto !== "" && montoGasto !== undefined) { expenses.push({ id: 'exp_' + rowNumber, name: conceptoGasto.toString().trim(), amount: Number(montoGasto) || 0, status: estado, sheetRow: rowNumber }); } if (conceptoIngreso && conceptoIngreso.toString().trim().toLowerCase() !== 'total' && montoIngreso !== "" && montoIngreso !== undefined) { incomes.push({ id: 'inc_' + rowNumber, name: conceptoIngreso.toString().trim(), amount: Number(montoIngreso) || 0, sheetRow: rowNumber }); } } return ContentService.createTextOutput(JSON.stringify({ incomes, expenses })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const contents = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const action = contents.action; const targetRow = Number(contents.sheetRow); const newAmount = Number(contents.amount) || 0; if (!targetRow || targetRow < 4) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Fila inválida' })) .setMimeType(ContentService.MimeType.JSON); } // Actualizar Gasto (Columna E) if (action === 'updateExpenseAmount') { sheet.getRange(targetRow, 5).setValue(newAmount); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } // Actualizar Ingreso (Columna G) if (action === 'updateIncomeAmount') { sheet.getRange(targetRow, 7).setValue(newAmount); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
